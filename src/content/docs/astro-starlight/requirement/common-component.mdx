## 共通コンポーネントの作成

ファイルは `src/components/CommonComponent.astro` とします。

```astro title="src/components/CommonComponent.astro"
---
import { LinkCard, Tabs, TabItem, Steps, FileTree } from "@astrojs/starlight/components";
import LinkPreview from "@components/LinkPreview.astro";
import MermaidChart from "@components/MermaidChart.astro";
import Table from "@components/Table.astro";

export const components = {
  LinkCard,
  Tabs,
  TabItem,
  Steps,
  FileTree,
  LinkPreview,
  MermaidChart
  Table,
};
---
```

### astro.config.mjs の設定

```js title="astro.config.mjs"
import { defineConfig } from "astro/config";
import starlight from "@astrojs/starlight";

export default defineConfig({
  integrations: [
    starlight({
      title: "My Customized Docs",
      components: {
        // デフォルトの Sidebar コンポーネントをカスタムコンポーネントに置き換え
        Sidebar: "./src/components/CustomSidebar.astro",
      },
    }),
  ],
});
```
